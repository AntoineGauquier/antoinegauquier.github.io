def recherche_dichotomique_efficace(liste_triee, x, gauche, droite):
    if gauche > droite:
            return -1
    milieu = int((gauche + droite)/2)

    if liste_triee[milieu] == x:
        return milieu

    elif liste_triee[milieu] > x:
        return recherche_dichotomique_efficace(liste_triee, x, gauche, milieu - 1)
    else:
        return recherche_dichotomique_efficace(liste_triee, x, milieu + 1, droite)

def recherche_dichotomique_copie(liste_triee, x):
    if len(liste_triee) == 0:
            return -1
    milieu = int(len(liste_triee) / 2)

    if liste_triee[milieu] == x:
        return milieu

    elif liste_triee[milieu] > x:
        sous_liste_triee = liste_triee[:milieu]
        return recherche_dichotomique_copie(sous_liste_triee, x)
    else:
        sous_liste_triee = liste_triee[milieu + 1:]
        resultat_sous_liste = recherche_dichotomique_copie(sous_liste_triee, x)
        if resultat_sous_liste == -1:
            return -1
        else:
            return resultat_sous_liste + milieu + 1

n = 100000
temps_dichotomie_efficace = []
temps_dichotomie_copie = []
for i in tqdm(range(n)):
    l = [i for i in range(i)]
    x = randint(-1, len(l))
    
    begin = perf_counter()
    recherche_dichotomique_efficace(l, x, 0, len(l) - 1)
    temps_dichotomie_efficace.append(perf_counter() - begin)

    begin = perf_counter()
    recherche_dichotomique_copie(l, x)
    temps_dichotomie_copie.append(perf_counter() - begin)
