def indice_minimum(A):
    if len(A) == 0:
        return None
    
    indice_minimum = 0
    minimum = A[0]
    
    for i in range(1, len(A)):
        if A[i] < minimum:
            minimum = A[i]
            indice_minimum = i
    
    return indice_minimum

def mesurer_temps(n):
    tailles = [100*i for i in range(0,n+1)]
    temps = []

    for taille in tqdm(tailles):
        tab = generer_tableau(taille)
        print(tab)
        debut = time()
        indice_minimum(tab)
        temps.append(time() - debut)

    return tailles, temps

def indice_minimum(liste):

    if len(A) == 0:
        return None

    indice_minimum = 0
    minimum = liste[0]

    for i in range(1, len(liste)): # Commencer à 0 marche aussi mais ça ne sert à rien
        if liste[i] < minimum:
            minimum = liste[i]
            indice_minimum = i

    return indice_minimum

    for taille in tailles:

        debut = time()
